\sectionBlock{Profile}{
\small
Ph.D. (CIFRE) candidate in Computer Science at Sorbonne University and EURECOM, in collaboration with Huawei Paris Research Center, working on efficient and reliable ultra-scale LLM and DNN training. I build symbolic performance and memory models, then couple them with ILP and constraint-based search to synthesize memory-feasible, communication-aware hybrid-parallel strategies and pipeline schedules. My work sits at the intersection of AI, HPC, and ML systems, has been validated on production Ascend clusters up to 16,384 NPUs, and has produced publications at CLUSTER, Euro-Par, NPC, and HPC Asia together with multiple patents and deployment-oriented planning workflows.
}

\sectionBlock{Research Focus}{
{\small Doctoral thesis: \textit{Systematic and Portable Optimisation of Hybrid Parallelism for Large-Scale Distributed Training} (Sorbonne University, doctoral school EDITE ED 130). A behaviour-level planning workflow built on a shared per-layer (Transformer) unit of reasoning, addressing one question: how can large-scale hybrid-parallel training be planned efficiently? Four capabilities, each realised as a system:}
\itemListStart
  \myItem{\textbf{Predictability --- PRISM}: a layer-level symbolic cost surface predicting compute, peak memory, and communication from model, training-config, and hardware descriptions, without profiling.}
  \myItem{\textbf{Interoperability --- BMPipe + H2O}: factor planning into an inner schedule/memory problem and an outer choice of parallelism degrees (data, tensor, pipeline, virtual-pipeline, sequence, expert, optimiser), optimised jointly via ILP/MILP.}
  \myItem{\textbf{Evolvability --- Concord}: a validity-preserving reuse contract that re-plans only the parts a model or configuration change invalidates, instead of restarting from scratch.}
  \myItem{\textbf{Adaptability --- ManuMatic}: runtime strategy injection that corrects planning decisions when observed execution diverges from the abstraction used to choose them.}
\itemListEnd
}

\sectionBlock{Selected Impact}{
\itemListStart
  \myItem{\textbf{BMPipe} (CLUSTER 2025): symbolic cost model plus ILP search over stage boundaries, intra-stage chunking, and per-layer recomputation; up to 1.36$\times$ speedup with 83--92\% HBM utilization, validated on clusters up to 16,384 Ascend NPUs.}
  \myItem{\textbf{ManuMatic} (NPC 2025): treated user-provided operator shardings as first-class planning constraints for robust ``manual + automatic'' search; 2.24$\times$ on Mixtral-8$\times$7B and 2.04$\times$ on Llama3-8B vs.\ D-Rec, and +30.1\% over an expert-tuned plan on Qwen2.5-72B.}
  \myItem{\textbf{H2O} (Euro-Par 2025, Best Poster): two-level workflow that searches parallelism degrees and micro-batch size, then refines layer-to-stage mapping and adaptive recomputation; MFU 26.13\%$\to$35.73\% and +36.72\% on DeepSeek-141B, transferred to production ($\approx$1.1$\times$ on 4,096 devices; up to 2.37$\times$ on long-sequence training at 512 devices).}
  \myItem{\textbf{PRISM} (HPC Asia 2026): profiling-free analytic planner predicting peak memory and communication; 1.23$\times$--1.43$\times$ speedups and MFU 29.40\%$\to$36.20\%, validated up to 1,024 devices.}
\itemListEnd
}
